-- M1. Catalogue/schéma : ceux fournis par le facilitateur. Données v1, EUR.
-- Seules les semaines avec attribution complète participent aux ratios.
-- Les dépenses exclues restent visibles. Ce ROAS n'est pas un effet causal.
SELECT channel,
  COUNT(*) AS weeks_total,
  SUM(attribution_complete) AS weeks_complete,
  SUM(spend_eur) AS spend_all_eur,
  SUM(CASE WHEN attribution_complete = 1 THEN spend_eur ELSE 0 END) AS spend_comparable_eur,
  SUM(CASE WHEN attribution_complete = 0 THEN spend_eur ELSE 0 END) AS spend_excluded_eur,
  SUM(CASE WHEN attribution_complete = 1 THEN new_customers ELSE 0 END) AS new_customers,
  SUM(CASE WHEN attribution_complete = 1 THEN attributed_net_revenue_eur ELSE 0 END) AS revenue_comparable_eur,
  ROUND(1.0 * SUM(CASE WHEN attribution_complete = 1 THEN attributed_net_revenue_eur ELSE 0 END)
    / NULLIF(SUM(CASE WHEN attribution_complete = 1 THEN spend_eur ELSE 0 END), 0), 4) AS attributed_roas,
  ROUND(1.0 * SUM(CASE WHEN attribution_complete = 1 THEN spend_eur ELSE 0 END)
    / NULLIF(SUM(CASE WHEN attribution_complete = 1 THEN new_customers ELSE 0 END), 0), 2) AS acquisition_cost_eur
FROM campaign_weekly
WHERE week_start >= '2026-09-21' AND week_start <= '2026-11-09'
GROUP BY channel
ORDER BY attributed_roas DESC, channel;

-- A1. Segmentation pédagogique de revue commerciale, pas de score ML.
-- Une règle lisible et explicite ; aucune campagne n'est envoyée.
WITH segments AS (
  SELECT account_id, segment, mrr_eur, email_opt_in, telemetry_complete,
    CASE WHEN telemetry_complete = 0 OR users_previous_28d <= 0 THEN 'Preuve insuffisante'
         WHEN 1.0 * users_current_28d / users_previous_28d >= 1.05 THEN 'Engagé'
         WHEN 1.0 * users_current_28d / users_previous_28d >= 0.60 THEN 'Stable'
         ELSE 'À revoir' END AS engagement_tier
  FROM account_snapshot WHERE as_of_date = '2026-11-17'
)
SELECT engagement_tier, COUNT(*) AS accounts, SUM(mrr_eur) AS mrr_eur,
  SUM(CASE WHEN email_opt_in = 1 AND telemetry_complete = 1
           AND engagement_tier <> 'Preuve insuffisante' THEN 1 ELSE 0 END) AS eligible_for_review
FROM segments
GROUP BY engagement_tier
ORDER BY engagement_tier;

-- S1. Pipeline ouvert vérifié de décembre ; pondération pédagogique non calibrée.
-- Une opportunité rejoint exactement un compte du snapshot.
SELECT a.country, COUNT(*) AS opportunities,
  SUM(o.amount_eur) AS open_pipeline_eur,
  ROUND(SUM(o.amount_eur * o.probability), 2) AS weighted_pipeline_eur,
  SUM(CASE WHEN o.days_since_activity >= 30 THEN o.amount_eur ELSE 0 END) AS stale_pipeline_eur
FROM opportunities o
JOIN account_snapshot a ON o.account_id = a.account_id
WHERE o.as_of_date = '2026-11-17' AND a.as_of_date = '2026-11-17'
  AND o.expected_close_date >= '2026-12-01' AND o.expected_close_date < '2027-01-01'
  AND o.stage IN ('Discovery','Evaluation','Negotiation')
  AND o.amount_verified = 1
GROUP BY a.country
ORDER BY weighted_pipeline_eur DESC, a.country;
